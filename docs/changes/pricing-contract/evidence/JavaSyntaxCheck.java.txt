import java.nio.file.*;
import java.util.*;
import javax.tools.*;
import com.sun.source.util.JavacTask;
public class JavaSyntaxCheck {
    public static void main(String[] args) throws Exception {
        var diagnostics=new DiagnosticCollector<JavaFileObject>();
        var compiler=ToolProvider.getSystemJavaCompiler();
        try(var files=compiler.getStandardFileManager(diagnostics,null,null);var stream=Files.walk(Path.of(args[0]))){
            var paths=stream.filter(p->p.toString().endsWith(".java")).toList();
            var task=(JavacTask)compiler.getTask(null,files,diagnostics,List.of("--release","21","-proc:none"),null,files.getJavaFileObjectsFromPaths(paths));
            int parsed=0; for(var unit:task.parse())parsed++;
            var errors=diagnostics.getDiagnostics().stream().filter(d->d.getKind()==Diagnostic.Kind.ERROR).toList();
            errors.forEach(System.out::println);
            System.out.println("PARSE ONLY: "+parsed+" Java files, "+errors.size()+" syntax errors. No symbol/type/framework validation.");
            if(!errors.isEmpty())System.exit(1);
        }
    }
}
